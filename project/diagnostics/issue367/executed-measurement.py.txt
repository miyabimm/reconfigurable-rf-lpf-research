# Wrappers observe the original implementation, leaving checks, budgets and storage order intact.
events=[]
def stamp():
    r=resource.getrusage(resource.RUSAGE_CHILDREN)
    return time.perf_counter(), time.process_time(), r.ru_utime+r.ru_stime
def wrap(obj, name, label):
    original=getattr(obj,name)
    def measured(*args, **kwargs):
        t=stamp()
        try: return original(*args, **kwargs)
        finally:
            q=stamp(); events.append(dict(name=label,wall=q[0]-t[0],cpu=q[1]-t[1],child_cpu=q[2]-t[2]))
    setattr(obj,name,measured)
wrap(ev.FixedStageEvaluation,'_contract','contract_base')
wrap(je,'source_identity','source_identity')
wrap(ev,'validate_values','validate_values')
wrap(RFEvaluator,'evaluate','rf_forward')
wrap(ev,'evaluate_requirements','requirements')
wrap(je.FixedJobEvaluation,'snapshot','snapshot')
# Measure atomic's whole operation and nested canonical, file/directory fsync and replace.
orig_atomic=storage.atomic

def atomic_measured(path, value):
    canonical, fsync, replace = storage.canonical, os.fsync, os.replace
    def instrument(fn,label):
        def f(*args,**kw):
            t=stamp()
            try: return fn(*args,**kw)
            finally:
                q=stamp(); events.append(dict(name=label,wall=q[0]-t[0],cpu=q[1]-t[1],child_cpu=q[2]-t[2]))
        return f
    storage.canonical=instrument(canonical,'atomic_canonical')
    os.fsync=instrument(fsync,'atomic_fsync')
    os.replace=instrument(replace,'atomic_replace')
    t=stamp()
    try: return orig_atomic(path,value)
    finally:
        q=stamp(); events.append(dict(name='atomic_total',wall=q[0]-t[0],cpu=q[1]-t[1],child_cpu=q[2]-t[2],bytes=Path(path).stat().st_size if Path(path).exists() else None))
        storage.canonical,os.fsync,os.replace=canonical,fsync,replace
ev.atomic=atomic_measured
